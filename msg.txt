#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

struct Student {
    string name;
    double height;
};

int main() {
    int T;
    cin >> T;

    vector<Student> students;
    string name;
    double height;

    // Read until q or Q
    while (cin >> name) {
        if (name == "q" || name == "Q")
            break;

        cin >> height;
        students.push_back({name, height});
    }

    // Copy heights and sort in descending order
    vector<double> heights;

    for (const auto& s : students)
        heights.push_back(s.height);

    sort(heights.begin(), heights.end(), greater<double>());

    // Height of the T-th tallest student
    double cutoff = heights[T - 1];

    // Print selected students in their original order
    for (const auto& s : students) {
        if (s.height >= cutoff) {
            cout << s.name << endl;
        }
    }

    return 0;
}
