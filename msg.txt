def checkVisa(x1, x2, y1, y2, z1, z2):
    if x1 >= x2 and y1 >= y2 and z1 <= z2:
        return "YES"
    else:
        return "NO"


import sys

print(checkVisa(
    int(sys.argv[1]),
    int(sys.argv[2]),
    int(sys.argv[3]),
    int(sys.argv[4]),
    int(sys.argv[5]),
    int(sys.argv[6])
), end='')
